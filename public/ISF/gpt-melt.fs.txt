#define PI       3.141592653589793
#define TWO_PI   (2.0 * PI)
const int OCTAVES = 4;

// 3D IQ-style random
float hash3(vec3 p) {
    p = fract(p * vec3(0.1031, 0.11369, 0.13787));
    p += dot(p, p.yzx + 19.19);
    return fract((p.x + p.y) * p.z);
}

// 3D smooth noise
float noise3(vec3 p) {
    vec3 i = floor(p);
    vec3 f = fract(p);
    vec3 u = f * f * (3.0 - 2.0 * f);
    float n000 = hash3(i + vec3(0.0));
    float n100 = hash3(i + vec3(1.0,0.0,0.0));
    float n010 = hash3(i + vec3(0.0,1.0,0.0));
    float n110 = hash3(i + vec3(1.0,1.0,0.0));
    float n001 = hash3(i + vec3(0.0,0.0,1.0));
    float n101 = hash3(i + vec3(1.0,0.0,1.0));
    float n011 = hash3(i + vec3(0.0,1.0,1.0));
    float n111 = hash3(i + vec3(1.0,1.0,1.0));
    float nx00 = mix(n000, n100, u.x);
    float nx10 = mix(n010, n110, u.x);
    float nx01 = mix(n001, n101, u.x);
    float nx11 = mix(n011, n111, u.x);
    float nxy0 = mix(nx00, nx10, u.y);
    float nxy1 = mix(nx01, nx11, u.y);
    return mix(nxy0, nxy1, u.z);
}

// 3D fractal Brownian motion
float fbm3(vec3 p) {
    float v = 0.0;
    float a = 0.6;
    for (int i = 0; i < OCTAVES; i++) {
        v += a * noise3(p);
        p *= 2.0;
        a *= 0.5;
    }
    return v;
}

// Hexagonal tiling function
vec2 hex(vec2 p) {
    float q = (p.x * 2.0 + p.y) / sqrt(3.0);
    float r = p.y * 2.0 / 3.0;
    vec2 id = floor(vec2(q, r));
    vec2 f = fract(vec2(q, r)) - 0.5;
    return f;
}

void main() {
    // normalized UV
    vec2 uv = gl_FragCoord.xy / RENDERSIZE.xy;
    vec2 centered = uv * 2.0 - 1.0;
    centered.x *= RENDERSIZE.x / RENDERSIZE.y;

    // build 3D position: X,Y from UV, Z from time
    float t = TIME * speed;
    vec3 pos = vec3(centered * scale, t * depthSpeed);

    // fluid 3D noise warp (melt)
    vec3 warp = pos + meltIntensity * vec3(
        fbm3(pos + vec3(1.0,0.0,0.0)),
        fbm3(pos + vec3(0.0,1.0,0.0)),
        fbm3(pos + vec3(0.0,0.0,1.0))
    );

    // project back to 2D
    vec2 wuv = warp.xy;

    // geometric tiling: hex grid
    vec2 h = hex(wuv);
    float grid = smoothstep(0.45, 0.48, length(h));

    // radial bloom mask
    float r = length(centered);
    float bloom = smoothstep(0.5, 0.0, r - 0.2 * sin(t * 0.5));

    // combine patterns
    float pattern = mix(grid, bloom, 0.6 + 0.4 * fbm3(warp * 0.5));

    // color by depth-based hue shift
    float hue = fract(fbm3(warp * 0.8) + t * 0.1);
    float sat = 0.8;
    float val = pattern;

    // HSV → RGB
    vec3 k = vec3(1.0, 2.0/3.0, 1.0/3.0);
    vec3 p1 = abs(fract(vec3(hue) + k.xyz) * 6.0 - k.www);
    vec3 rgb = val * mix(k.xxx, clamp(p1 - k.xxx, 0.0, 1.0), sat);

    // output with slight vignette
    float d = length(centered);
    float vig = smoothstep(0.7, 1.0, d);
    gl_FragColor = vec4(rgb * (1.0 - vig), 1.0);
}
